import pandas as pd
import os

def create_comparison_excel(file_a, file_b, output_path):
    # 1. Check if input files exist
    if not os.path.exists(file_a) or not os.path.exists(file_b):
        print("Error: Input files not found.")
        return

    print("Reading files... this may take a moment.")

    try:
        # Read all sheets into dictionaries
        xls_a = pd.read_excel(file_a, sheet_name=None)
        xls_b = pd.read_excel(file_b, sheet_name=None)
    except Exception as e:
        print(f"Error reading Excel files: {e}")
        return

    try:
        # Use xlsxwriter for stability
        with pd.ExcelWriter(output_path, engine='xlsxwriter') as writer:
            
            # Iterate through EVERY sheet in Client A
            for sheet_name, df_a in xls_a.items():
                
                # Create a copy so we don't modify the original
                current_df = df_a.copy()
                current_df['Source_File'] = 'Client A'

                # Check if Client B has this specific page
                if sheet_name in xls_b:
                    df_b = xls_b[sheet_name].copy()
                    df_b['Source_File'] = 'Client B'
                    
                    # Stack Client B data underneath Client A
                    current_df = pd.concat(
                        [current_df, df_b],
                        ignore_index=True
                    )
                    status = "Merged with Client B"
                else:
                    status = "Client A Only (No match in B)"

                # Write to the comparison file
                current_df.to_excel(
                    writer, 
                    sheet_name=sheet_name, 
                    index=False
                )
                
                print(f"Sheet '{sheet_name}': {status}")

            print("-" * 40)
            print(f"Done! Saved to '{output_path}'")

    except PermissionError:
        print(f"Error: Close '{output_path}' before running this.")
    except Exception as e:
        print(f"An unexpected error occurred: {e}")

# --- CONFIGURATION ---
client_a_file = 'client_a.xlsx'
client_b_file = 'client_b.xlsx'
output_file = 'comparison.xlsx'

if __name__ == "__main__":
    create_comparison_excel(
        client_a_file, 
        client_b_file, 
        output_file
    )
